    14→/* ============ SCROLL PROGRESS ============ */
    15→function ScrollProgress() {
    16→  const [pct, setPct] = React.useState(0);
    17→  React.useEffect(() => {
    18→    const onScroll = () => {
    19→      const d = document.documentElement;
    20→      const h = d.scrollHeight - d.clientHeight;
    21→      setPct(h > 0 ? Math.min(100, (d.scrollTop / h) * 100) : 0);
    22→    };
    23→    onScroll();
    24→    window.addEventListener("scroll", onScroll, { passive: true });
    25→    return () => window.removeEventListener("scroll", onScroll);
    26→  }, []);
    27→  return (
    28→    <div className="pointer-events-none fixed inset-x-0 top-0 z-[55] h-[3px]" aria-hidden="true">
    29→      <div className="h-full bg-gradient-to-r from-rust via-brass to-rust shadow-[0_0_8px_rgba(181,80,46,0.6)] transition-[width] duration-75" style={{ width: `${pct}%` }} />
    30→    </div>
    31→  );
    32→}
