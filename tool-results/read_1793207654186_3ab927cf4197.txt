    14→/* ============ SCROLL PROGRESS ============ */
    15→function ScrollProgress() {
    16→  const [pct, setPct] = React.useState(0);
    17→  React.useEffect(() => {
    18→    const onScroll = () => {
    19→      const d = document.documentElement;
    20→      const h = d.scrollHeight - d.clientHeight;
    21→      setPct(h > 0 ? Math.min(100, (d.scrollTop / h) * 100) : 0);
    22→    };
    23→    onScroll();
    24→    window.addEventListener("scroll", onScroll, { passive: true });
    25→    return () => window.removeEventListener("scroll", onScroll);
    26→  }, []);
    27→  return (
    28→    <div className="pointer-events-none fixed inset-x-0 top-0 z-[55] h-[3px]" aria-hidden="true">
    29→      <div className="h-full bg-gradient-to-r from-rust via-brass to-rust shadow-[0_0_8px_rgba(181,80,46,0.6)] transition-[width] duration-75" style={{ width: `${pct}%` }} />
    30→    </div>
    31→  );
    32→}

   686→/* ============ COOKIE BAR ============ */
   687→const COOKIE_KEY = "stal-cookie-ok";
   688→
   689→function CookieBar({ onOpenPrivacy }: { onOpenPrivacy: () => void }) {
   690→  const { t } = useI18n();
   691→  const [show, setShow] = React.useState(false);
   692→
   693→  React.useEffect(() => {
   694→    if (typeof window === "undefined") return;
   695→    let dismissed = false;
   696→    try { dismissed = window.localStorage.getItem(COOKIE_KEY) === "1"; } catch { /* приватный режим — просто покажем плашку */ }
   697→    if (dismissed) return;
   698→    const timer = window.setTimeout(() => setShow(true), 900);
   699→    return () => window.clearTimeout(timer);
   700→  }, []);
   701→
   702→  const dismiss = () => {
   703→    setShow(false);
   704→    try { window.localStorage.setItem(COOKIE_KEY, "1"); } catch { /* ок, просто закрываем */ }
   705→  };
   706→
   707→  return (
   708→    <div
   709→      className={`fixed bottom-4 left-4 right-4 z-[60] border border-ash/30 bg-ink-2/95 p-5 shadow-[0_20px_50px_-15px_rgba(0,0,0,0.7)] backdrop-blur-md transition-all duration-500 sm:bottom-5 sm:right-auto sm:max-w-sm ${show ? "translate-y-0 opacity-100" : "pointer-events-none translate-y-4 opacity-0"}`}
   710→      role="region"
   711→      aria-label="Cookies"
   712→      aria-hidden={!show}
   713→    >
   714→      <span className="pointer-events-none absolute left-1.5 top-1.5 h-2.5 w-2.5 border-l border-t border-brass/50" />
   715→      <span className="pointer-events-none absolute right-1.5 top-1.5 h-2.5 w-2.5 border-r border-t border-brass/50" />
   716→      <div className="flex items-center gap-2 font-display text-[0.62rem] uppercase tracking-[0.3em] text-brass"><span className="bulb-dot" />Cookies</div>
   717→      <p className="mt-3 text-[0.82rem] leading-relaxed text-cream-dim">{t.footer.cookieText}</p>
   718→      <div className="mt-4 flex items-center justify-between gap-3">
   719→        <button type="button" onClick={onOpenPrivacy} className="font-display text-[0.68rem] uppercase tracking-[0.16em] text-cream-dim underline decoration-ash/40 underline-offset-4 transition-colors hover:text-brass hover:decoration-brass">{t.footer.privacy}</button>
   720→        <button type="button" onClick={dismiss} className="btn-rust px-4 py-2.5 text-xs">{t.footer.cookieOk}</button>
   721→      </div>
   722→    </div>
   723→  );
   724→}

   739→/* ============ FLOATING BOOK CTA ============ */
   740→function FloatingBook() {
   741→  const { t } = useI18n();
   742→  const [show, setShow] = React.useState(false);
   743→
   744→  React.useEffect(() => {
   745→    const hero = document.getElementById("top");
   746→    const booking = document.getElementById("booking");
   747→    if (!hero || !booking) return;
   748→
   749→    let heroVisible = true;
   750→    let bookingVisible = false;
   751→    const update = () => setShow(!heroVisible && !bookingVisible);
   752→
   753→    // Hero: срезаем 15% сверху и снизу — кнопка появляется, когда первый экран почти прошёл
   754→    const heroObs = new IntersectionObserver(([e]) => { heroVisible = e.isIntersecting; update(); },
   755→      { rootMargin: "-15% 0px -15% 0px" });
   756→    // Booking: считаем видимым, когда блок реально вошёл в экран (не просто показался краем)
   757→    const bookingObs = new IntersectionObserver(([e]) => { bookingVisible = e.isIntersecting; update(); },
   758→      { rootMargin: "0px 0px -30% 0px" });
   759→
   760→    heroObs.observe(hero);
   761→    bookingObs.observe(booking);
   762→    return () => { heroObs.disconnect(); bookingObs.disconnect(); };
   763→  }, []);
   764→
   765→  return (
   766→    <button
   767→      type="button"
   768→      onClick={() => document.getElementById("booking-card")?.scrollIntoView({ behavior: "smooth", block: "center" })}
   769→      className={`cta-float group fixed bottom-[5.5rem] right-5 z-40 flex items-center gap-2.5 border border-rust bg-rust px-4 py-3 font-display text-xs font-semibold uppercase tracking-[0.16em] text-cream transition-all duration-300 hover:bg-[#c75e38] hover:shadow-[0_0_0_1px_#a9793a,0_10px_28px_-8px_rgba(181,80,46,0.75)] hover:-translate-y-0.5 active:translate-y-0 sm:px-5 sm:py-3.5 sm:text-sm lg:flex ${show ? "translate-y-0 opacity-100" : "pointer-events-none translate-y-4 opacity-0"}`}
   770→      style={{ clipPath: "polygon(10px 0, 100% 0, 100% calc(100% - 10px), calc(100% - 10px) 100%, 0 100%, 0 10px)" }}
   771→      aria-label={t.ui.bookNow}
   772→      aria-hidden={!show}
   773→      tabIndex={show ? 0 : -1}
   774→    >
   775→      <span className="relative flex h-2 w-2 shrink-0" aria-hidden="true">
   776→        <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-cream/70" />
   777→        <span className="relative inline-flex h-2 w-2 rounded-full bg-cream" />
   778→      </span>
   779→      <span className="hidden sm:inline">{t.ui.bookNow}</span>
   780→      <span className="sm:hidden">{t.ui.book}</span>
   781→      <ArrowDownIcon className="hidden h-4 w-4 transition-transform duration-300 group-hover:translate-y-0.5 sm:block" />
   782→    </button>
   783→  );
   784→}
